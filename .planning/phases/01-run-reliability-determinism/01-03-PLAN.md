```python
def _handle_rerun(run_id_str: str) -> None:
    """Rerun a previous campaign by retrieving stored request and config.
    
    D-07: Use CLI flag --rerun <run_id> to replay execution
    D-08: Create child run_id, preserve parent linkage
    """
    from src.runtime.run_store import RunStore
    from src.models.campaign_models import RunID, CampaignRequest
    from src.workflow.crew_workflow import CampaignCrew
    
    store = RunStore()
    
    # Retrieve original run's config_snapshot and parent_run_id
    try:
        run_id = RunID(value=run_id_str)  # Reconstruct RunID from string
    except ValueError as e:
        console.print(f"[red]Invalid run_id format: {run_id_str}[/red]")
        console.print("[dim]Expected format: YYYYMMDDTHHMMSS-XXXXX[/dim]")
        sys.exit(1)
    
    metadata = store.get_run(run_id)
    if not metadata:
        console.print(f"[red]Run not found: {run_id}[/red]")
        sys.exit(1)
    
    # D-07: Recreate CampaignRequest from config_snapshot
    request = CampaignRequest.model_validate(metadata.config_snapshot)
    console.print(f"[dim]Replaying campaign from run {run_id}[/dim]")
    console.print(f"[dim]Product: {request.product_name}[/dim]")
    
    # D-08: Create new child run_id with parent linkage
    crew = CampaignCrew(request, store=store, parent_run_id=run_id)
    
    console.print(f"[dim]New child run ID: {crew.run_id}[/dim]")
    
    try:
        brief = crew.run()
        console.print("[green]Rerun completed successfully[/green]")
    except Exception as e:
        console.print(f"[red]Rerun failed: {_format_error_message(e, debug=os.getenv('DEBUG_MODE') == 'true')}[/red]")
        sys.exit(1)

def _format_error_message(exc: Exception, debug: bool) -> str:
    """Format error for display based on debug mode (D-09, D-10).
    
    Normal mode: User-safe error code and suggestion
    Debug mode: Full exception trace
    """
    if debug:
        return ''.join(traceback.format_exception(type(exc), exc, exc.__traceback__))
    else:
        # Sanitize: provide stable error code + recovery suggestion
        error_code = _classify_error_type(exc).__class__.__name__
        if "rate" in str(exc).lower():
            return f"[{error_code}] Request rate limited. Try again in a few minutes."
        elif "api" in str(exc).lower():
            return f"[{error_code}] Provider API error. Check your API key and quota."
        elif "network" in str(exc).lower():
            return f"[{error_code}] Network error. Check your connection and try again."
        else:
            return f"[{error_code}] Operation failed. Run with --debug for details."
```

```python
# D-05, D-06: Typed error classification
def _classify_error_type(self, exc: Exception) -> str:
    """Classify error as transient (retryable) or hard (fail fast).
    
    Transient errors: rate_limit, network timeout, temporary upstream failure
    Hard errors: validation, auth, configuration
    
    Returns: error type string for logging/diagnostics
    """
    error_str = str(exc).lower()
    exc_type = type(exc).__name__.lower()
    
    # Rate limit (already handled in main retry loop, but classify for posterity)
    if any(marker in error_str for marker in ["rate limit", "rate_limit_exceeded", "429", "too many requests"]):
        return "TRANSIENT_RATE_LIMIT"
    
    # Network timeouts
    if any(marker in error_str for marker in ["timeout", "connection refused", "connection reset"]):
        return "TRANSIENT_NETWORK"
    
    # Temporary upstream (provider temporary unavailability)
    if any(marker in error_str for marker in ["service unavailable", "503", "500", "temporarily"]):
        return "TRANSIENT_UPSTREAM"
    
    # Hard errors: validation, auth, config
    if any(marker in error_str for marker in ["validationerror", "pydantic", "invalid input"]):
        return "HARD_VALIDATION"
    
    if any(marker in error_str for marker in ["unauthorized", "forbidden", "401", "403", "apikey", "auth"]):
        return "HARD_AUTH"
    
    if any(marker in error_str for marker in ["configurationerror", "missing", "not found"]):
        return "HARD_CONFIG"
    
    # Default to hard error (fail fast unless proven transient)
    return "HARD_UNKNOWN"

def _format_error_message(self, exc: Exception) -> str:
    """Present error to user based on debug mode setting (D-09, D-10).
    
    Normal mode (settings.debug_mode=false):
      - User-safe, stable error code
      - Suggested recovery action
      - No internal trace
    
    Debug mode (settings.debug_mode=true):
      - Full exception traceback
      - Internal state details
      - Provider payload (if available)
    """
    error_type = self._classify_error_type(exc)
    
    if settings.debug_mode:
        # D-10: Full diagnostics
        import traceback
        trace = ''.join(traceback.format_exception(type(exc), exc, exc.__traceback__))
        return f"[ERROR {error_type}] {trace}\nRun ID: {self.run_id}"
    else:
        # D-09: Sanitized, user-safe message
        suggestions = {
            "TRANSIENT_RATE_LIMIT": "Rate limit exceeded. Wait a few minutes and use --rerun to retry.",
            "TRANSIENT_NETWORK": "Network error. Check your connection and use --rerun to retry.",
            "TRANSIENT_UPSTREAM": "Provider temporarily unavailable. Use --rerun to retry.",
            "HARD_VALIDATION": "Invalid input or configuration. Check campaign request parameters.",
            "HARD_AUTH": "Authentication failed. Check your API credentials.",
            "HARD_CONFIG": "Configuration missing. Check environment variables and settings.",
            "HARD_UNKNOWN": "Operation failed. Run with --debug for details.",
        }
        suggestion = suggestions.get(error_type, "Unknown error. Run with --debug for details.")
        return f"[{error_type}] {suggestion}"
```

| Threat ID | Category | Component | Disposition | Mitigation Plan |
