def _save_outputs(self, brief: CampaignBrief) -> None:
    # (existing artifact saving code)
    md_path = self.output_path / f"{self.request.product_name.replace(' ', '_')}_{self.run_id}.md"
    json_path = self.output_path / f"{self.request.product_name.replace(' ', '_')}_{self.run_id}.json"
    
    md_path.write_text(md_content)
